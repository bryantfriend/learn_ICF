async function checkTutorial(page) {
  const address = page.url().split('#')[0].split('?')[0];
  const base = address.slice(0, address.lastIndexOf('/') + 1);
  const errors = [];
  page.on('pageerror', function record(error) { errors.push(error.message); });
  function assert(value, message) { if (!value) throw new Error(message); }
  async function open(fragment) {
    await page.goto(base + 'index.html' + fragment);
    await page.locator('#tutorialNav .group-toggle').first().waitFor({state:'attached'});
    await page.waitForFunction(function ready() { return document.querySelector('#view h1') && document.querySelector('#tutorialNav a'); });
  }

  await open('#overview');
  await page.setViewportSize({ width: 1440, height: 950 });
  await page.getByRole('heading', { name: 'ICF Wiki', exact: true }).waitFor();
  assert(await page.locator('.brand img').evaluate(function loaded(img) { return img.complete && img.naturalWidth > 0; }), 'Brand image failed to load');
  assert(await page.locator('.stage-card').count() === 6, 'Missing stage link');
  await page.getByRole('button', { name: 'Run example' }).click();
  await page.waitForFunction(function ran() { return document.querySelector('.pg-example-status').textContent.includes('advanced-project'); });
  const catalog = await page.evaluate(async function enumerate() {
    const module = await import('./tutorial-lessons.mjs');
    const course = await import('./academy-course.mjs');
    const readings = module.chapters.flatMap(function flatten(chapter) { return chapter.lessons; });
    const ids = new Set(readings.map(function id(lesson) { return lesson.id; }));
    return { lessons: readings, unique: ids.size, valid: readings.every(function valid(lesson) { return course.lessonById.has(lesson.practice); }) };
  });
  assert(catalog.lessons.length === catalog.unique && catalog.valid, 'Lesson IDs or exercise references are invalid');
  for (const item of catalog.lessons) {
    await open('#lesson/' + item.id);
    await page.getByRole('heading', { level: 1, name: item.title, exact: true }).waitFor();
    assert(await page.locator('.lesson-section').count() === item.sections.length, 'Section missing from ' + item.id);
    await page.locator('#quiz input').nth(item.question.answer).check();
    await page.getByRole('button', { name:'Check answer', exact:true }).click();
    assert(await page.locator('#quiz .correct').isVisible(), 'Answer mismatch for ' + item.id);
  }
  await open('#overview');
  await page.getByLabel('Filter wiki pages').fill('createNormalizePayloadNumberFieldNormalizer');
  await page.locator('#tutorialNav a[href="#helper/normalize-createNormalizePayloadNumberFieldNormalizer"]').click();
  await page.getByRole('heading',{level:1,name:'createNormalizePayloadNumberFieldNormalizer'}).waitFor();
  await page.getByLabel('Filter wiki pages').fill('');
  const wikiCount = await page.evaluate(async function visitWiki() {
    const wiki = await import('./tutorial-wiki.mjs');
    const entries = [];
    for (const e of wiki.examples) entries.push(['#example/'+e.id,e.title]);
    for (const e of wiki.apiEntries) entries.push(['#helper/'+e.id,e.key]);
    entries.push(['#setup','Set up and use offline'],['#build','Build a complete feature']);
    for (const [hash,title] of entries) {
      const changed = new Promise(function listen(resolve) { window.addEventListener('hashchange',resolve,{once:true}); });
      location.hash = hash;
      await changed;
      if (document.querySelector('#view h1').textContent !== title) throw new Error('Incorrect wiki page at '+hash);
      if (!document.querySelector('#tutorialNav a[aria-current="page"]')) throw new Error('Page missing from sidebar at '+hash);
    }
    return entries.length;
  });
  await page.keyboard.press('/');
  const input = page.getByRole('searchbox',{name:'Search wiki topics, examples, and API entries'});
  await input.fill('Adaptive Context');
  assert(await page.locator('#searchResults a').count() > 0, 'Search missed Adaptive Context');
  await page.keyboard.press('Escape');
  assert(!await page.locator('#searchDialog').isVisible(), 'Escape must dismiss search with a query');
  await page.keyboard.press('/');
  await input.fill('Pipeline playground');
  await page.keyboard.press('Enter');
  await page.getByRole('heading', { level: 1, name: 'Pipeline playground' }).waitFor();
  await page.getByLabel('Scenario',{exact:true}).selectOption('denied');
  await page.waitForFunction(function denied() { return document.querySelector('.pg-trace-status').textContent.includes('Stopped at Authorize'); });
  await page.getByLabel('Scenario',{exact:true}).selectOption('review');
  await page.waitForFunction(function review() { return document.querySelector('.pg-trace-status').textContent.includes('prerequisite-review'); });
  await page.setViewportSize({ width:320, height:800 });
  await open('#overview');
  const menu = page.getByRole('button', {name:'Open wiki menu'});
  await menu.click();
  const last = page.locator('#sidebar a').last();
  await last.focus();
  await page.keyboard.press('Tab');
  assert(await menu.evaluate(function focused(button) { return button === document.activeElement; }), 'Focus escaped the mobile menu');
  await page.setViewportSize({ width:1000,height:800 });
  await page.waitForFunction(function drawerClosed() { return !document.body.classList.contains('menu-open') && !document.getElementById('reader').inert; });
  for (const fragment of ['#overview','#curriculum','#lesson/first-run','#exercises','#reference','#playground']) {
    await page.setViewportSize({ width:320,height:800 });
    await open(fragment);
    assert(await page.evaluate(function fits() { return document.documentElement.scrollWidth <= innerWidth; }), 'Page overflow on ' + fragment);
  }
  assert(errors.length === 0, errors.join('\n'));
  return 'Passed ' + catalog.unique + ' topics and ' + wikiCount + ' example, API and implementation pages, sidebar filtering, practice links, quiz answers, the live example, playground, keyboard navigation and mobile layout.';
}
